# 1️⃣ Check current branch and file status
git status

# 2️⃣ Add files to staging area (add all files)
git add .

# 3️⃣ Commit the changes locally
git commit -m "Your commit message here"

# 4️⃣ Fetch latest changes from remote repository
git fetch origin

# 5️⃣ Merge remote branch changes into your current branch
git merge origin/branch-1-15

# 6️⃣ Push your committed code to remote repository
git push origin branch-1-15
